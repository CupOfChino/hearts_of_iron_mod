#============== 宣战事件：清算旧盟 ==============
country_event = { #宣战旧盟友
	id = postwar.3
	title = postwar.3.t
	desc = postwar.3.d

	picture = GFX_report_event_generic_political

	is_triggered_only = yes

	option = { #宣战旧盟友，成为世界霸主
		name = postwar.3.a
		ai_chance = {
			factor = 100
		}
		every_country = {
			limit = {
				has_country_flag = postwar_former_ally_flag
				NOT = { tag = ROOT }
				NOT = { is_in_faction_with = ROOT }
				NOT = { has_war_with = ROOT }
			}
			ROOT = {
				declare_war_on = { 
					type = annex_everything
					target = PREV 
				}
			}
		}
	}
}

#============== 组阵营事件：三皇同盟凯旋线 ==============
# 由分裂效果（postwar_axis_split_effect）末尾立即触发，
# 统一为德、法、匈、中创建战后阵营并依次播报新闻（全局 flag 防重复，只执行一次）
country_event = { #组阵营：三皇同盟凯旋线
	id = postwar.4
	title = postwar.4.t
	desc = postwar.4.d

	picture = GFX_report_event_generic_political

	is_triggered_only = yes

	option = { #各霸一方，问鼎世界！
		name = postwar.4.a
		ai_chance = {
			factor = 100
		}
		if = {
			limit = {
				country_exists = GER
				GER = {
					is_subject = no
				}
			}
			GER = {
				set_rule = { can_create_factions = yes }
				create_faction_from_template = the_mod_faction_GER_kaiserbund_faction
				set_country_flag = postwar_debug_faction_created_flag
				set_country_flag = now_has_a_faction_flag
			}
			news_event = { id = change_news.80 days = 1 }
		}

		if = {
			limit = {
				country_exists = FRA
				FRA = {
					is_subject = no
				}
			}
			FRA = {
				set_rule = { can_create_factions = yes }
				create_faction_from_template = the_mod_faction_FRA_continental_system_faction
				set_country_flag = postwar_debug_faction_created_flag
				set_country_flag = now_has_a_faction_flag
			}
			news_event = { id = change_news.81 days = 1 }
		}

		if = {
			limit = {
				country_exists = HUN
				HUN = {
					is_subject = no
				}
			}
			HUN = {
				set_rule = { can_create_factions = yes }
				create_faction_from_template = the_mod_faction_HUN_habsburg_alliance_faction
				set_country_flag = postwar_debug_faction_created_flag
				set_country_flag = now_has_a_faction_flag
				
			}
			news_event = { id = change_news.82 days = 1 }
		}

		if = {
			limit = {
				country_exists = CHI
				CHI = {
					is_subject = no
				}
			}
			CHI = {
				set_rule = { can_create_factions = yes }
				create_faction_from_template = the_mod_faction_CHI_celestial_empire_faction

		}

	}
