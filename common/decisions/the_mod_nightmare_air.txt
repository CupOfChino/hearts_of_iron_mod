					instant_build = yes

			if = {
				limit = { date < 1942.1.1 }
				if = {
					limit = { NOT = { has_country_flag = NIGHTMARE_GER_f2_design } }

						name = "Bf 109 K"
						type = fighter_equipment_2
						upgrades = {
							plane_gun_upgrade = 2
							plane_engine_upgrade = 1
							plane_range_upgrade = 1
							plane_reliability_upgrade = 2
						}
					}
					set_country_flag = NIGHTMARE_GER_f2_design
				}
				if = {
					limit = { has_global_flag = NIGHTMARE_air_mode_swarm }
					add_equipment_to_stockpile = {
						type = fighter_equipment_2
						amount = 1600
						variant_name = "Bf 109 K"
					}
					add_equipment_to_stockpile = {
						type = CAS_equipment_2
						amount = 800
					}
					add_equipment_to_stockpile = {
						type = naval_bomber_equipment_2
						amount = 400
					}
				}
				else = {
					add_equipment_to_stockpile = {
						type = fighter_equipment_2
						amount = 800
						variant_name = "Bf 109 K"
					}
					add_equipment_to_stockpile = {
						type = CAS_equipment_2
						amount = 400
					}
					add_equipment_to_stockpile = {
						type = naval_bomber_equipment_2
						amount = 200
					}
				}
			}
			else = {
				if = {
					limit = { NOT = { has_country_flag = NIGHTMARE_GER_f3_design } }
					create_equipment_variant = {
						name = "Fw 190 D"
						type = fighter_equipment_3

							plane_gun_upgrade = 2
							plane_engine_upgrade = 1
							plane_range_upgrade = 1
